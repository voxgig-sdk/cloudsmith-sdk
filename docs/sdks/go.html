Set up the client

This setup fragment reads credentials from the environment. Use the API operation reference to supply input for each call.

package main

import (
    "os"
    sdk "github.com/voxgig-sdk/cloudsmith-sdk/go"
)

func main() {
    client := sdk.NewCloudsmithSDK(map[string]any{"apikey": os.Getenv("CLOUDSMITH_APIKEY")})
    _ = client
}
